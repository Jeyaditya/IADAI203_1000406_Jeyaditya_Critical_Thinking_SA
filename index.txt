  <script>
    let currentMood = 2;
    let currentST = 4.5;
    let timerInterval = null;
    let secondsRemaining = 15 * 60;

    function goToScreen(screenId) {
      document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
      document.getElementById(screenId).classList.add('active');

      const nav = document.getElementById('bottomNav');
      if (screenId === 'screen-splash' || screenId === 'screen-mood') {
        nav.classList.remove('visible');
      } else {
        nav.classList.add('visible');
      }

      document.querySelectorAll('.nav-item').forEach(btn => {
        btn.classList.toggle('active', btn.dataset.target === screenId);
      });
    }

    function updateMoodUI(val) {
      currentMood = parseInt(val);
      const moods = {
        1: ['😫', '1 / 5 — Deeply Burned Out'],
        2: ['😔', '2 / 5 — Overwhelmed & Tired'],
        3: ['😐', '3 / 5 — Neutral / Mentally Foggy'],
        4: ['🙂', '4 / 5 — Calm & Focused'],
        5: ['🌿', '5 / 5 — Grounded & Recharged']
      };
      document.getElementById('moodEmoji').textContent = moods[currentMood][0];
      document.getElementById('moodText').textContent = moods[currentMood][1];
      recalculateAI();
    }

    function toggleTag(el) {
      el.classList.toggle('selected');
    }

    function saveMoodAndProceed() {
      const emojis = {1:'😫', 2:'😔', 3:'😐', 4:'🙂', 5:'🌿'};
      document.getElementById('dashMood').textContent = `${emojis[currentMood]} ${currentMood}/5`;
      recalculateAI();
      goToScreen('screen-home');
    }

    function updateScreenTime(val) {
      currentST = parseFloat(val);
      document.getElementById('stValueLabel').textContent = `${currentST} hours`;
      const hrs = Math.floor(currentST);
      const mins = Math.round((currentST - hrs) * 60);
      document.getElementById('dashST').textContent = `${hrs}h ${mins}m`;
      recalculateAI();
    }

    function recalculateAI() {
      const ruleTag = document.getElementById('aiRuleTag');
      const title = document.getElementById('aiPlanTitle');
      const desc = document.getElementById('aiPlanDesc');
      const homeHeadline = document.getElementById('homeAiHeadline');

      if (currentMood <= 2 && currentST > 2.5) {
        ruleTag.textContent = 'HIGH-STRESS INTERVENTION';
        title.textContent = '15-Min Analog Eye & Breath Reset';
        homeHeadline.textContent = '15-Min Analog Eye & Mind Reset';
        desc.innerHTML = '1. Place phone face-down across the room.<br>2. Practice 4-4-4-4 box breathing for 3 minutes.<br>3. Stretch shoulders & rest eyes from blue light.';
        secondsRemaining = 15 * 60;
      } else if (currentMood >= 4 && currentST <= 2.5) {
        ruleTag.textContent = 'POSITIVE REINFORCEMENT';
        title.textContent = '5-Min Mindful Gratitude Check-In';
        homeHeadline.textContent = '5-Min Mindful Gratitude Check-In';
        desc.innerHTML = '1. Great balance today! Screen time is well within healthy limits.<br>2. Log one win in your Gratitude Journal to lock in your Calm Streak.';
        secondsRemaining = 5 * 60;
      } else {
        ruleTag.textContent = 'MICRO-BREAK POMODORO';
        title.textContent = '10-Min 20-20-20 Eye & Stretch Break';
        homeHeadline.textContent = '10-Min Eye & Hydration Break';
        desc.innerHTML = '1. Look at an object 20 feet away for 20 seconds.<br>2. Drink a full glass of water away from screens.<br>3. Take a short walk before resuming study.';
        secondsRemaining = 10 * 60;
      }
      updateTimerDisplay();
    }

    function updateTimerDisplay() {
      const m = String(Math.floor(secondsRemaining / 60)).padStart(2, '0');
      const s = String(secondsRemaining % 60).padStart(2, '0');
      document.getElementById('timerDisplay').textContent = `${m}:${s}`;
    }

    function toggleTimer() {
      const btn = document.getElementById('timerBtn');
      if (timerInterval) {
        clearInterval(timerInterval);
        timerInterval = null;
        btn.textContent = '▶ Resume Screen-Free Timer';
      } else {
        btn.textContent = '⏸ Pause Offline Timer';
        timerInterval = setInterval(() => {
          if (secondsRemaining > 0) {
            secondsRemaining--;
            updateTimerDisplay();
          } else {
            clearInterval(timerInterval);
            btn.textContent = '✅ Reset Completed! (+1 Streak)';
          }
        }, 1000);
      }
    }

    function addJournalEntry() {
      const input = document.getElementById('journalInput');
      const text = input.value.trim();
      if (!text) return;
      const feed = document.getElementById('journalFeed');
      const div = document.createElement('div');
      div.className = 'journal-entry';
      div.innerHTML = `<strong>Just Now • 10:15 PM (Mindful ✨)</strong><br>${text}`;
      feed.prepend(div);
      input.value = '';
    }

    function openExplainModal() {
      document.getElementById('modalMood').textContent = `${currentMood}/5`;
      document.getElementById('modalST').textContent = `${currentST} hrs`;
      document.getElementById('explainModal').classList.add('open');
    }

    function closeExplainModal() {
      document.getElementById('explainModal').classList.remove('open');
    }

    function triggerPushNudge() {
      const banner = document.getElementById('pushBanner');
      banner.classList.add('show');
      setTimeout(() => banner.classList.remove('show'), 6500);
    }

    function acceptPushNudge() {
      document.getElementById('pushBanner').classList.remove('show');
      goToScreen('screen-ai');
    }
